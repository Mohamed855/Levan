// Exercises only the submitted mock HTTP server. Does not launch a browser or contact Laravel.
import fs from 'node:fs';
import path from 'node:path';
import {createRequire} from 'node:module';
const require = createRequire(path.resolve('package.json'));
const source = fs.readFileSync('docs/qa/evidence/site-auth/puppeteer_suite.cjs', 'utf8');
const end = source.indexOf('const screenSizes =');
if (end < 0) throw new Error('Submitted server definition not found');
const serverSource = source.slice(0, end).replace("const puppeteer = require('puppeteer');", '');
const {server} = new Function('require', '__dirname', serverSource + '\nreturn {server};')(
    require, path.resolve('docs/qa/evidence/site-auth'));
await new Promise(resolve => server.listen(0, '127.0.0.1', resolve));
try {
    const origin = `http://127.0.0.1:${server.address().port}`;
    const validation = await fetch(origin + '/superadmin/form-validation', {
        method: 'POST', headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({code: 'not-a-totp', fields: ['code']}),
    });
    const challenge = await fetch(origin + '/superadmin/two-factor-challenge', {
        method: 'POST', redirect: 'manual', headers: {'Content-Type': 'application/x-www-form-urlencoded'},
        body: 'code=not-a-totp',
    });
    const multipart = new FormData();
    multipart.set('code', '000000');
    const multipartValidation = await fetch(origin + '/superadmin/form-validation', {
        method: 'POST', body: multipart,
    });
    const dashboard = await fetch(origin + '/dashboard');
    console.log(JSON.stringify({
        scope: 'Submitted Node harness only, not production Laravel behavior',
        invalidNonNumericCodeValidationStatus: validation.status,
        invalidCodeChallengeWithoutPendingOrCsrfStatus: challenge.status,
        multipartWrongCodeValidationStatus: multipartValidation.status,
        syntheticAuthCookieIssued: Boolean(challenge.headers.get('set-cookie')),
        dashboardWithoutCookieStatus: dashboard.status,
        realLaravelSessionTotpOrGuardVerified: false,
    }, null, 2));
} finally {
    await new Promise(resolve => server.close(resolve));
}
